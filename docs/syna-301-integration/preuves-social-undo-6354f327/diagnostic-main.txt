import {afterEach,beforeEach,expect,it,vi} from 'vitest';
import {account,project} from './conversation209-fixtures';
import {all,id,now,run} from '@/lib/db';
import {loadProject} from '@/lib/projects';
import {applySocialEdit,editPostV2} from '@/lib/social-v2/engine';
import {mockSocialDeps} from './social-v2-mock';
const at=Date.parse('2026-11-05T08:00:00Z');
beforeEach(()=>{vi.spyOn(Date,'now').mockReturnValue(at);vi.stubGlobal('fetch',vi.fn(()=>{throw Error('NO_PROVIDER');}));});afterEach(()=>{expect(fetch).not.toHaveBeenCalled();vi.unstubAllGlobals();vi.restoreAllMocks();});
async function fixture(){const u=await account(),pid=project(u,'products'),p=loadProject(pid),postId=id();run("INSERT INTO posts(id,project_id,network,format,status,title,caption,media,publish_key,engine,scheduled_at,created_at,updated_at) VALUES(?,?,'facebook','text','review','Source fictive','TEXTE_AVANT','[]',?,'v2',?,?,?)",postId,pid,`fixture:${postId}`,Date.parse('2026-11-10T10:00:00Z'),now(),now());const {deps,log}=mockSocialDeps({aiActive:false}),duplicated=await applySocialEdit(p,'Duplique les publications',deps,{confirm:true,at});expect(duplicated.applied).toBe(true);const copies=all<{id:string}>('SELECT id FROM posts WHERE project_id=? AND id!=?',pid,postId);expect(copies).toHaveLength(1);return{pid,p,postId,copyId:copies[0].id,deps,log};}
it('control: native undo removes an untouched copy and leaves the source',async()=>{const f=await fixture(),r=await applySocialEdit(f.p,'Annule ma dernière modification',f.deps,{confirm:true,at});expect(r.applied).toBe(true);expect(all('SELECT id,caption FROM posts WHERE project_id=?',f.pid)).toEqual([{id:f.postId,caption:'TEXTE_AVANT'}]);expect(Object.values(f.log).every(v=>v===0)).toBe(true);});
it('a copy edited natively at the same timestamp must not be deleted by undo',async()=>{const f=await fixture();editPostV2(f.p,f.copyId,{caption:'COPIE_MODIFIEE_MANUELLEMENT'});await applySocialEdit(f.p,'Annule ma dernière modification',f.deps,{confirm:true,at});expect(all('SELECT id,caption FROM posts WHERE id=?',f.copyId)).toEqual([{id:f.copyId,caption:'COPIE_MODIFIEE_MANUELLEMENT'}]);expect(Object.values(f.log).every(v=>v===0)).toBe(true);});
it('a source edited natively at the same timestamp must not be overwritten by undo',async()=>{const f=await fixture();editPostV2(f.p,f.postId,{caption:'SOURCE_MODIFIEE_MANUELLEMENT'});await applySocialEdit(f.p,'Annule ma dernière modification',f.deps,{confirm:true,at});expect(all('SELECT id,caption FROM posts WHERE id=?',f.postId)).toEqual([{id:f.postId,caption:'SOURCE_MODIFIEE_MANUELLEMENT'}]);expect(Object.values(f.log).every(v=>v===0)).toBe(true);});
